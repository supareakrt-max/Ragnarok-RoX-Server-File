//===== rAthena Script =======================================
//= Ro-X Player AI - เมนู @ai (ขั้นที่ 1-6)
//===== คำอธิบาย ============================================
//= - @ai        เปิดหน้าต่างตั้งค่า AI
//= - @ai on     เปิด AI ทันที
//= - @ai off    ปิด AI ทันที
//= - ต้องใช้ map-server.exe ที่คอมไพล์รวม src/custom/ai_player.inc แล้ว
//=   ถ้ายังไม่ได้คอมไพล์ ไฟล์นี้จะ error ตอนโหลด
//= - ค่าตั้งเก็บในตัวแปรตัวละคร (ดูรายชื่อใน ai_player.inc)
//=   AI_RANGE AI_NOBOSS AI_SKILL[] AI_SKSP AI_BUFF[] AI_HPPCT AI_HPITEM AI_SPPCT AI_SPITEM
//=   AI_LOOT AI_LOOTID[] AI_RT_HP AI_RT_POT AI_RT_WEIGHT AI_FLYWING
//=   ต่อแมพ: AI_MM_<แมพ> (โหมดเลือกมอน), AI_ML_<แมพ>$ (รายการมอนที่ติ๊ก)
//=   ซัพพอร์ต: AI_ROLE AI_LEADER AI_LEADERN$ AI_FOLLOW AI_HEALSK[0] AI_HEALPCT AI_PBUFF[] AI_RES
//= - เวลาใช้งาน AI: #AI_TIME (วินาที, ต่อไอดี) นับถอยหลังเฉพาะตอน AI ทำงาน
//=   ซื้อ 4 ชม. = 1,000,000 Zeny หรือ 2 Cash Point (แก้ราคาได้ที่ OnInit)
//=   โหมดซัพพอร์ต ใช้ฟรี ไม่หักเวลา
//= - ไอเทม AI Controller (.item_id) กดใช้ = เปิดเมนูนี้ (ไม่หาย, ทิ้ง/ขาย/ใส่รถเข็นไม่ได้)
//=   ต้องเพิ่มไอเทมใน db/import/item_db.yml ก่อน (ดู ai_system/db/item_db_ai.yml)
//= - ไฟล์นี้บันทึกเป็น TIS-620
//============================================================

-	script	AI_Menu	-1,{
OnInit:
	bindatcmd "ai", strnpcinfo(3) + "::OnAtcmd", 0, 99;
	.n$ = "[ ระบบ AI ]";
	setarray .mode$[0], "ตีทุกตัว", "ตีเฉพาะที่ติ๊ก", "ตีทุกตัว ยกเว้นที่ติ๊ก";
	setarray .loot$[0], "ไม่เก็บ", "เก็บทุกชิ้น", "เก็บเฉพาะที่เลือก";
	setarray .role$[0], "ฟาร์มมอน", "ซัพพอร์ตเดินตาม";
	// ----- ราคาเวลาใช้งาน AI -----
	.buy_hours = 4;			// ซื้อครั้งละกี่ชั่วโมง
	.buy_zeny = 1000000;		// ราคา Zeny
	.buy_cash = 2;			// ราคา Cash Point (#CASHPOINTS)
	.max_hours = 720;		// เก็บเวลาสะสมได้สูงสุด (ชม.)
	.gm_level = 99;			// GM เลเวลนี้ขึ้นไป เติมเวลาฟรีได้
	// ----- ไอเทมเปิดเมนู AI -----
	.item_id = 60400;		// AI Controller
	.give_item = 1;			// 1 = แจกตัวละครละ 1 ชิ้นตอนล็อกอินครั้งแรก
	end;

// แจกไอเทม AI Controller ตัวละครละ 1 ครั้ง
OnPCLoginEvent:
	if (.give_item && !AI_ITEMGIVEN && getitemname(.item_id) != "null") {
		AI_ITEMGIVEN = 1;
		if (!countitem(.item_id))
			getitem .item_id, 1;
	}
	end;

// กดใช้ไอเทม AI Controller (item script: doevent "AI_Menu::OnItem";)
OnItem:
	goto L_Main;

OnAtcmd:
	.@arg$ = strtolower(.@atcmd_parameters$[0]);
	if (.@arg$ == "on") {
		callsub S_On, 0;
		end;
	}
	if (.@arg$ == "off") {
		dispbottom (ai_off() ? "[AI] ปิด AI แล้ว" : "[AI] AI ปิดอยู่แล้ว");
		end;
	}

L_Main:
	if (.@shown)
		clear;
	.@shown = 1;
	mes .n$;
	mes "สถานะ: " + (ai_status() ? "^00AA00เปิดอยู่^000000" : "^FF0000ปิดอยู่^000000") + "   โหมด: ^0055FF" + .role$[AI_ROLE] + "^000000";
	if (AI_ROLE == 1)
		mes "เวลา AI: ^00AA00ฟรี (โหมดซัพพอร์ต)^000000";
	else
		mes "เวลา AI เหลือ: ^0055FF" + callsub(S_TimeStr, #AI_TIME) + "^000000";
	if (AI_ROLE == 1)
		mes "ตาม: ^0055FF" + (AI_LEADER ? AI_LEADERN$ : "- ยังไม่เลือก -") + "^000000";
	else
		mes "แมพ " + strcharinfo(3) + ": ^0055FF" + .mode$[getd("AI_MM_" + strcharinfo(3))] + "^000000";
	switch (select((ai_status() ? "ปิด AI" : "เปิด AI") + ":ซื้อเวลา AI:เปลี่ยนโหมด (ฟาร์ม / ซัพพอร์ต):การโจมตี / เลือกมอน:สกิลโจมตี:ตั้งค่าซัพพอร์ต (ตาม / ฮีล / บัฟปาร์ตี้):บัฟตัวเอง:ยา HP / SP:เก็บของ:กลับเมือง / ปีกแมลงวัน:ปิดหน้าต่าง")) {
	case 1:
		clear;
		mes .n$;
		if (ai_status()) {
			ai_off();
			mes "ปิด AI แล้ว";
			close;
		}
		callsub S_On, 1;
		close;
	case 2: callsub S_Buy; break;
	case 3:
		AI_ROLE = !AI_ROLE;
		if (ai_status()) {
			clear;
			mes .n$;
			mes "เปลี่ยนเป็นโหมด ^0055FF" + .role$[AI_ROLE] + "^000000 แล้ว";
			mes "(มีผลทันที)";
			next;
		}
		break;
	case 4: callsub S_Attack; break;
	case 5: callsub S_Skill; break;
	case 6: callsub S_Support; break;
	case 7: callsub S_Buff; break;
	case 8: callsub S_Potion; break;
	case 9: callsub S_Loot; break;
	case 10: callsub S_Home; break;
	default: close;
	}
	goto L_Main;

// ---------------------------------------------------------------
// getarg(0) = วินาที -> "X ชม. Y นาที"
S_TimeStr:
	.@t = getarg(0);
	if (.@t <= 0)
		return "หมดแล้ว";
	return (.@t / 3600) + " ชม. " + ((.@t % 3600) / 60) + " นาที";

// ---------------------------------------------------------------
S_Buy:
	clear;
	mes .n$;
	mes "เวลา AI เหลือ: ^0055FF" + callsub(S_TimeStr, #AI_TIME) + "^000000";
	mes "(ใช้ได้ทุกตัวละครในไอดี นับเฉพาะตอน AI ทำงาน)";
	mes "^00AA00โหมดซัพพอร์ต ใช้ฟรี ไม่ต้องซื้อเวลา^000000";
	mes "ซื้อ " + .buy_hours + " ชม.:";
	mes "- " + callfunc("F_InsertComma", .buy_zeny) + " Zeny  (มี " + callfunc("F_InsertComma", Zeny) + ")";
	mes "- " + .buy_cash + " Cash Point  (มี " + #CASHPOINTS + ")";
	.@menu$ = "จ่าย " + callfunc("F_InsertComma", .buy_zeny) + " Zeny:จ่าย " + .buy_cash + " Cash Point:" + (getgmlevel() >= .gm_level ? "[GM] เติมฟรี 24 ชม." : "") + ":กลับ";
	.@s = select(.@menu$);
	if (.@s == 4)
		return;
	.@add = (.@s == 3 ? 24 : .buy_hours) * 3600;
	if (#AI_TIME + .@add > .max_hours * 3600) {
		clear;
		mes .n$;
		mes "เก็บเวลาสะสมได้ไม่เกิน " + .max_hours + " ชม.";
		next;
		return;
	}
	clear;
	mes .n$;
	if (.@s == 1) {
		if (Zeny < .buy_zeny) {
			mes "Zeny ไม่พอ";
			next;
			return;
		}
		Zeny -= .buy_zeny;
	} else if (.@s == 2) {
		if (#CASHPOINTS < .buy_cash) {
			mes "Cash Point ไม่พอ";
			next;
			return;
		}
		#CASHPOINTS -= .buy_cash;
	}
	#AI_TIME += .@add;
	logmes "AI time +" + (.@add / 3600) + "h (" + (.@s == 1 ? "zeny" : (.@s == 2 ? "cash" : "gm")) + ")";
	mes "เพิ่มเวลา AI แล้ว";
	mes "เหลือ: ^0055FF" + callsub(S_TimeStr, #AI_TIME) + "^000000";
	next;
	return;

// ---------------------------------------------------------------
S_Support:
	while (1) {
		clear;
		mes .n$;
		mes "ตาม: ^0055FF" + (AI_LEADER ? AI_LEADERN$ : "- ยังไม่เลือก -") + "^000000  ระยะ " + (AI_FOLLOW ? AI_FOLLOW : 3) + " ช่อง";
		mes "ฮีล: " + (AI_HEALSK[0] ? ai_skillname(AI_HEALSK[0]) + " เมื่อ HP < " + (AI_HEALPCT ? AI_HEALPCT : 70) + "%" : "ไม่ฮีล");
		mes "ชุบชีวิต: " + (AI_RES ? "เปิด" : "ปิด");
		mes "บัฟปาร์ตี้ (รวม Magnificat / Gloria):";
		callsub S_ShowSkills, "AI_PBUFF", 6;
		switch (select("เลือกคนที่จะตาม (ในปาร์ตี้):ตั้งระยะตาม:เลือกสกิลฮีล:ตั้ง % HP ที่จะฮีล:เพิ่มบัฟปาร์ตี้:ลบบัฟปาร์ตี้:" + (AI_RES ? "ปิด" : "เปิด") + " ชุบชีวิต:กลับ")) {
		case 1:
			if (!getcharid(1)) {
				clear;
				mes .n$;
				mes "ต้องอยู่ในปาร์ตี้ก่อน";
				next;
				break;
			}
			getpartymember getcharid(1), 0;
			getpartymember getcharid(1), 1;
			.@c = 0;
			.@menu$ = "";
			for (.@i = 0; .@i < $@partymembercount; .@i++) {
				if ($@partymembercid[.@i] == getcharid(0))
					continue;
				.@cids[.@c] = $@partymembercid[.@i];
				.@names$[.@c] = $@partymembername$[.@i];
				.@c++;
				.@menu$ += $@partymembername$[.@i] + ":";
			}
			clear;
			mes .n$;
			if (.@c == 0) {
				mes "ไม่มีสมาชิกคนอื่นในปาร์ตี้";
				next;
				break;
			}
			mes "เลือกคนที่จะเดินตาม";
			.@s = select(.@menu$ + "ยกเลิก") - 1;
			if (.@s < .@c) {
				AI_LEADER = .@cids[.@s];
				AI_LEADERN$ = .@names$[.@s];
			}
			break;
		case 2:
			clear;
			mes .n$;
			mes "เดินตามห่างกี่ช่อง (1-10)";
			if (input(.@v, 1, 10) == 0)
				AI_FOLLOW = .@v;
			break;
		case 3:
			AI_HEALSK[0] = 0;
			callsub S_PickSkill, 3, "AI_HEALSK", 1;
			break;
		case 4:
			clear;
			mes .n$;
			mes "ฮีลเมื่อ HP ต่ำกว่ากี่ % (1-99)";
			if (input(.@v, 1, 99) == 0)
				AI_HEALPCT = .@v;
			break;
		case 5: callsub S_PickSkill, 4, "AI_PBUFF", 6; break;
		case 6: callsub S_RemoveSkill, "AI_PBUFF", 6; break;
		case 7: AI_RES = !AI_RES; break;
		default: return;
		}
	}

// ---------------------------------------------------------------
// เปิด AI  getarg(0): 1 = อยู่ในหน้าต่าง (mes), 0 = จากคำสั่ง (dispbottom)
S_On:
	if (AI_ROLE == 1 && !AI_LEADER)
		.@r = -5;
	else
		.@r = ai_on();
	switch (.@r) {
	case 1:  .@m$ = "[AI] เปิด AI แล้ว"; break;
	case -1: .@m$ = "[AI] ใช้ AI ในเมืองไม่ได้"; break;
	case -2: .@m$ = "[AI] ตัวละครเสียชีวิตอยู่"; break;
	case -3: .@m$ = "[AI] น้ำหนักเกิน 90% ใช้ AI ไม่ได้"; break;
	case -4: .@m$ = "[AI] ไม่มีเวลา AI เหลือ - ซื้อได้ที่ @ai > ซื้อเวลา AI"; break;
	case -5: .@m$ = "[AI] โหมดซัพพอร์ต: เลือกคนที่จะตามก่อน (@ai > ตั้งค่าซัพพอร์ต)"; break;
	default: .@m$ = "[AI] เปิด AI ไม่สำเร็จ"; break;
	}
	if (getarg(0)) {
		mes .@m$;
		if (.@r == 1) {
			if (AI_ROLE == 1)
				mes "โหมดซัพพอร์ต ใช้ฟรี";
			else
				mes "เวลา AI เหลือ: " + callsub(S_TimeStr, #AI_TIME);
			mes "ปิดหน้าต่างนี้เพื่อเริ่มทำงาน";
		}
	} else
		dispbottom .@m$;
	return;

// ---------------------------------------------------------------
S_Attack:
	while (1) {
		.@map$ = strcharinfo(3);
		clear;
		mes .n$;
		mes "รัศมีหามอน: ^0055FF" + (AI_RANGE ? AI_RANGE : 9) + "^000000 ช่อง";
		mes "ตีบอส / MVP: " + (AI_NOBOSS ? "^FF0000ไม่ตี^000000" : "^00AA00ตี^000000");
		mes "มอนในแมพนี้: ^0055FF" + .mode$[getd("AI_MM_" + .@map$)] + "^000000";
		switch (select("ตั้งรัศมีหามอน:" + (AI_NOBOSS ? "เปลี่ยนเป็น ตีบอส/MVP" : "เปลี่ยนเป็น ไม่ตีบอส/MVP") + ":เลือกมอนที่จะตี (แมพนี้):กลับ")) {
		case 1:
			clear;
			mes .n$;
			mes "ใส่รัศมีหามอน (3-14 ช่อง)";
			if (input(.@v, 3, 14) == 0)
				AI_RANGE = .@v;
			break;
		case 2:
			AI_NOBOSS = !AI_NOBOSS;
			break;
		case 3:
			callsub S_MobPick;
			break;
		default:
			return;
		}
	}

// ---------------------------------------------------------------
// ติ๊กเลือกมอนจากรายชื่อมอนในแมพ (จำแยกตามแมพ)
S_MobPick:
	.@map$ = strcharinfo(3);
	while (1) {
		.@mode = getd("AI_MM_" + .@map$);
		.@list$ = getd("AI_ML_" + .@map$ + "$");
		.@n = ai_maplist();
		clear;
		mes .n$;
		mes "แมพ: " + .@map$ + "  โหมด: ^0055FF" + .mode$[.@mode] + "^000000";
		if (.@mode == 0)
			mes "^FF0000โหมด ตีทุกตัว ติ๊กไว้ก็ไม่มีผล^000000";
		mes "[/] = ติ๊กแล้ว   [  ] = ยังไม่ติ๊ก";
		if (.@n == 0)
			mes "ตอนนี้ไม่มีมอนในแมพนี้";
		.@menu$ = "เปลี่ยนโหมด:ติ๊กทั้งหมด:เอาออกทั้งหมด";
		for (.@i = 0; .@i < .@n; .@i++) {
			.@id = @ai_mob_id[.@i];
			.@b$ = "";
			if (@ai_mob_boss[.@i] == 2)
				.@b$ = " [MVP]";
			else if (@ai_mob_boss[.@i] == 1)
				.@b$ = " [บอส]";
			.@menu$ += ":" + (compare(.@list$, "," + .@id + ",") ? "[/] " : "[  ] ") + getmonsterinfo(.@id, MOB_NAME) + .@b$ + " (" + @ai_mob_cnt[.@i] + " ตัว)";
		}
		.@s = select(.@menu$ + ":กลับ");
		if (.@s == 1) {
			setd "AI_MM_" + .@map$, (.@mode + 1) % 3;
			continue;
		}
		if (.@s == 2) {
			.@list$ = ",";
			for (.@i = 0; .@i < .@n; .@i++)
				.@list$ += @ai_mob_id[.@i] + ",";
			setd "AI_ML_" + .@map$ + "$", (.@n ? .@list$ : "");
			continue;
		}
		if (.@s == 3) {
			setd "AI_ML_" + .@map$ + "$", "";
			continue;
		}
		.@k = .@s - 4;
		if (.@k >= .@n)
			return;
		.@id = @ai_mob_id[.@k];
		if (compare(.@list$, "," + .@id + ","))
			.@list$ = replacestr(.@list$, "," + .@id + ",", ",");
		else
			.@list$ = (.@list$ == "" ? "," : .@list$) + .@id + ",";
		if (.@list$ == ",")
			.@list$ = "";
		setd "AI_ML_" + .@map$ + "$", .@list$;
	}

// ---------------------------------------------------------------
S_Skill:
	while (1) {
		clear;
		mes .n$;
		mes "สกิลโจมตี (ใช้ตามลำดับ บนลงล่าง):";
		callsub S_ShowSkills, "AI_SKILL", 5;
		mes "ใช้สกิลเมื่อ SP >= ^0055FF" + AI_SKSP + "%^000000";
		mes "(สกิลใช้ไม่ได้ / SP ไม่พอ = ตีธรรมดา)";
		switch (select("เพิ่มสกิล:ลบสกิล:ตั้ง SP ขั้นต่ำ:กลับ")) {
		case 1: callsub S_PickSkill, 1, "AI_SKILL", 5; break;
		case 2: callsub S_RemoveSkill, "AI_SKILL", 5; break;
		case 3:
			clear;
			mes .n$;
			mes "ใช้สกิลโจมตีเมื่อ SP เหลืออย่างน้อยกี่ % (0-100)";
			if (input(.@v, 0, 100) == 0)
				AI_SKSP = .@v;
			break;
		default:
			return;
		}
	}

S_Buff:
	while (1) {
		clear;
		mes .n$;
		mes "บัฟตัวเอง (ร่ายใหม่เมื่อหมด):";
		callsub S_ShowSkills, "AI_BUFF", 6;
		switch (select("เพิ่มบัฟ:ลบบัฟ:กลับ")) {
		case 1: callsub S_PickSkill, 2, "AI_BUFF", 6; break;
		case 2: callsub S_RemoveSkill, "AI_BUFF", 6; break;
		default: return;
		}
	}

// getarg(0) = ชื่อตัวแปร, getarg(1) = จำนวนช่อง
S_ShowSkills:
	.@c = 0;
	for (.@i = 0; .@i < getarg(1); .@i++) {
		.@id = getd(getarg(0) + "[" + .@i + "]");
		if (.@id) {
			.@c++;
			mes .@c + ". " + ai_skillname(.@id) + " Lv" + getskilllv(.@id);
		}
	}
	if (.@c == 0)
		mes "- ยังไม่ได้เลือก";
	return;

// getarg(0) = ชนิด (1 โจมตี, 2 บัฟตัวเอง, 3 ฮีล, 4 บัฟปาร์ตี้), getarg(1) = ชื่อตัวแปร, getarg(2) = จำนวนช่อง
S_PickSkill:
	.@type = getarg(0);
	.@var$ = getarg(1);
	.@max = getarg(2);
	for (.@slot = 0; .@slot < .@max; .@slot++)
		if (!getd(.@var$ + "[" + .@slot + "]"))
			break;
	clear;
	mes .n$;
	if (.@slot >= .@max) {
		mes "เต็มแล้ว (สูงสุด " + .@max + " สกิล) ลบออกก่อน";
		next;
		return;
	}
	getskilllist;
	.@c = 0;
	.@menu$ = "";
	for (.@i = 0; .@i < @skilllist_count; .@i++) {
		.@id = @skilllist_id[.@i];
		if (@skilllist_lv[.@i] < 1)
			continue;
		.@t = ai_skilltype(.@id);
		// บัฟตัวเองเลือกบัฟที่ใส่คนอื่นได้ด้วย / บัฟปาร์ตี้เลือกสกิลร่ายใส่ตัวที่มีผลทั้งปาร์ตี้ได้ (Magnificat, Gloria ...)
		if (.@t != .@type && !(.@type == 2 && .@t == 4) && !(.@type == 4 && .@t == 2))
			continue;
		for (.@j = 0; .@j < .@max; .@j++)
			if (getd(.@var$ + "[" + .@j + "]") == .@id)
				break;
		if (.@j < .@max)
			continue;	// เลือกไว้แล้ว
		.@ids[.@c] = .@id;
		.@c++;
		.@menu$ += ai_skillname(.@id) + " Lv" + @skilllist_lv[.@i] + ":";
	}
	if (.@c == 0) {
		mes "ไม่มีสกิลที่เลือกได้";
		next;
		return;
	}
	mes "เลือกสกิล (ใช้เลเวลสูงสุดที่มี)";
	.@s = select(.@menu$ + "ยกเลิก") - 1;
	if (.@s < .@c)
		setd .@var$ + "[" + .@slot + "]", .@ids[.@s];
	return;

// getarg(0) = ชื่อตัวแปร, getarg(1) = จำนวนช่อง
S_RemoveSkill:
	.@var$ = getarg(0);
	.@max = getarg(1);
	.@c = 0;
	.@menu$ = "";
	for (.@i = 0; .@i < .@max; .@i++) {
		.@id = getd(.@var$ + "[" + .@i + "]");
		if (!.@id)
			continue;
		.@slots[.@c] = .@i;
		.@c++;
		.@menu$ += ai_skillname(.@id) + ":";
	}
	clear;
	mes .n$;
	if (.@c == 0) {
		mes "ยังไม่ได้เลือกสกิล";
		next;
		return;
	}
	mes "เลือกสกิลที่จะลบ";
	.@s = select(.@menu$ + "ยกเลิก") - 1;
	if (.@s >= .@c)
		return;
	setd .@var$ + "[" + .@slots[.@s] + "]", 0;
	// จัดเรียงใหม่ ไม่ให้มีช่องว่าง
	.@k = 0;
	for (.@i = 0; .@i < .@max; .@i++) {
		.@id = getd(.@var$ + "[" + .@i + "]");
		setd .@var$ + "[" + .@i + "]", 0;
		if (.@id) {
			.@tmp[.@k] = .@id;
			.@k++;
		}
	}
	for (.@i = 0; .@i < .@k; .@i++)
		setd .@var$ + "[" + .@i + "]", .@tmp[.@i];
	return;

S_Potion:
	while (1) {
		clear;
		mes .n$;
		if (AI_HPITEM && AI_HPPCT)
			mes "HP: ใช้ ^0055FF" + getitemname(AI_HPITEM) + "^000000 เมื่อ HP < " + AI_HPPCT + "%";
		else
			mes "HP: ไม่ใช้ยา";
		if (AI_SPITEM && AI_SPPCT)
			mes "SP: ใช้ ^0055FF" + getitemname(AI_SPITEM) + "^000000 เมื่อ SP < " + AI_SPPCT + "%";
		else
			mes "SP: ไม่ใช้ยา";
		switch (select("เลือกยา HP:ตั้ง % HP:เลือกยา SP:ตั้ง % SP:ไม่ใช้ยา HP:ไม่ใช้ยา SP:กลับ")) {
		case 1:
			.@it = callsub(S_PickInv, 1);
			if (.@it) {
				AI_HPITEM = .@it;
				if (!AI_HPPCT)
					AI_HPPCT = 50;
			}
			break;
		case 2:
			clear;
			mes .n$;
			mes "กินยา HP เมื่อ HP ต่ำกว่ากี่ % (1-99)";
			if (input(.@v, 1, 99) == 0)
				AI_HPPCT = .@v;
			break;
		case 3:
			.@it = callsub(S_PickInv, 1);
			if (.@it) {
				AI_SPITEM = .@it;
				if (!AI_SPPCT)
					AI_SPPCT = 30;
			}
			break;
		case 4:
			clear;
			mes .n$;
			mes "กินยา SP เมื่อ SP ต่ำกว่ากี่ % (1-99)";
			if (input(.@v, 1, 99) == 0)
				AI_SPPCT = .@v;
			break;
		case 5:
			AI_HPITEM = 0;
			AI_HPPCT = 0;
			break;
		case 6:
			AI_SPITEM = 0;
			AI_SPPCT = 0;
			break;
		default:
			return;
		}
	}

// เลือกไอเทมจากกระเป๋า  getarg(0): 1 = เฉพาะยา (Healing), 0 = ทุกชนิด  -> คืนค่า item id หรือ 0
S_PickInv:
	getinventorylist;
	.@c = 0;
	.@menu$ = "";
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		.@id = @inventorylist_id[.@i];
		if (getarg(0) && getiteminfo(.@id, ITEMINFO_TYPE) != IT_HEALING)
			continue;
		for (.@j = 0; .@j < .@c; .@j++)
			if (.@ids[.@j] == .@id)
				break;
		if (.@j < .@c)
			continue;
		.@ids[.@c] = .@id;
		.@c++;
		.@menu$ += getitemname(.@id) + " (" + countitem(.@id) + "):";
	}
	clear;
	mes .n$;
	if (.@c == 0) {
		mes (getarg(0) ? "ไม่มียาในกระเป๋า" : "ไม่มีไอเทมในกระเป๋า");
		next;
		return 0;
	}
	mes "เลือกไอเทมจากกระเป๋า";
	.@s = select(.@menu$ + "ยกเลิก") - 1;
	if (.@s >= .@c)
		return 0;
	return .@ids[.@s];

// ---------------------------------------------------------------
S_Loot:
	while (1) {
		clear;
		mes .n$;
		mes "เก็บของ: ^0055FF" + .loot$[AI_LOOT] + "^000000";
		.@c = 0;
		for (.@i = 0; .@i < 20; .@i++) {
			if (AI_LOOTID[.@i]) {
				.@c++;
				if (.@c <= 10)
					mes "- " + getitemname(AI_LOOTID[.@i]);
			}
		}
		if (.@c > 10)
			mes "... และอีก " + (.@c - 10) + " ชิ้น";
		if (.@c == 0)
			mes "(ยังไม่มีรายการไอเทม)";
		if (AI_LOOT == 2 && .@c == 0)
			mes "^FF0000โหมดนี้ต้องเพิ่มไอเทมก่อน^000000";
		switch (select("เปลี่ยนโหมด:เพิ่มไอเทมจากกระเป๋า:เพิ่มด้วยเลข ID:ลบไอเทม:ล้างรายการ:กลับ")) {
		case 1:
			AI_LOOT = (AI_LOOT + 1) % 3;
			break;
		case 2:
			.@it = callsub(S_PickInv, 0);
			if (.@it)
				callsub S_AddLoot, .@it;
			break;
		case 3:
			clear;
			mes .n$;
			mes "ใส่เลข ID ไอเทม";
			input .@it;
			if (.@it > 0 && getitemname(.@it) != "null")
				callsub S_AddLoot, .@it;
			else {
				clear;
				mes .n$;
				mes "ไม่มีไอเทม ID นี้";
				next;
			}
			break;
		case 4:
			.@menu$ = "";
			.@k = 0;
			for (.@i = 0; .@i < 20; .@i++) {
				if (!AI_LOOTID[.@i])
					continue;
				.@slots[.@k] = .@i;
				.@k++;
				.@menu$ += getitemname(AI_LOOTID[.@i]) + ":";
			}
			if (.@k == 0)
				break;
			clear;
			mes .n$;
			mes "เลือกไอเทมที่จะลบ";
			.@s = select(.@menu$ + "ยกเลิก") - 1;
			if (.@s < .@k)
				AI_LOOTID[.@slots[.@s]] = 0;
			break;
		case 5:
			deletearray AI_LOOTID;
			break;
		default:
			return;
		}
	}

S_AddLoot:
	.@it = getarg(0);
	.@free = -1;
	for (.@i = 0; .@i < 20; .@i++) {
		if (AI_LOOTID[.@i] == .@it)
			return;	// มีแล้ว
		if (.@free < 0 && !AI_LOOTID[.@i])
			.@free = .@i;
	}
	if (.@free < 0) {
		clear;
		mes .n$;
		mes "รายการเต็มแล้ว (สูงสุด 20 ชิ้น)";
		next;
		return;
	}
	AI_LOOTID[.@free] = .@it;
	return;

S_Home:
	while (1) {
		clear;
		mes .n$;
		mes "กลับเมือง (ใช้ ^0055FFButterfly Wing^000000 ในกระเป๋า):";
		mes "- HP ต่ำกว่า: " + (AI_RT_HP ? AI_RT_HP + "%" : "ปิด");
		mes "- ยา HP หมด: " + ((AI_RT_POT & 1) ? "กลับ" : "ไม่กลับ");
		mes "- ยา SP หมด: " + ((AI_RT_POT & 2) ? "กลับ" : "ไม่กลับ");
		mes "- น้ำหนักถึง: " + (AI_RT_WEIGHT ? AI_RT_WEIGHT + "%" : "ปิด");
		mes "ปีกแมลงวันเมื่อหามอนไม่เจอ: " + (AI_FLYWING ? AI_FLYWING + " วินาที" : "ปิด");
		mes "(กลับเมืองแล้ว AI จะปิดเอง)";
		switch (select("ตั้ง HP ต่ำ:สลับ ยา HP หมด:สลับ ยา SP หมด:ตั้งน้ำหนัก:ตั้งปีกแมลงวัน:กลับ")) {
		case 1:
			clear;
			mes .n$;
			mes "กลับเมืองเมื่อ HP ต่ำกว่ากี่ % (0 = ปิด, 1-99)";
			if (input(.@v, 0, 99) == 0)
				AI_RT_HP = .@v;
			break;
		case 2:
			AI_RT_POT = AI_RT_POT ^ 1;
			break;
		case 3:
			AI_RT_POT = AI_RT_POT ^ 2;
			break;
		case 4:
			clear;
			mes .n$;
			mes "กลับเมืองเมื่อน้ำหนักถึงกี่ % (0 = ปิด, 50-89)";
			if (input(.@v, 0, 89) == 0 && (.@v == 0 || .@v >= 50))
				AI_RT_WEIGHT = .@v;
			break;
		case 5:
			clear;
			mes .n$;
			mes "ใช้ Fly Wing เมื่อหามอนไม่เจอกี่วินาที (0 = ปิด, 5-120)";
			if (input(.@v, 0, 120) == 0 && (.@v == 0 || .@v >= 5))
				AI_FLYWING = .@v;
			break;
		default:
			return;
		}
	}
