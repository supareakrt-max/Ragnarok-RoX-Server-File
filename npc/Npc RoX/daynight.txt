//===== rAthena Script =======================================
//= Npc RoX : กลางวัน / กลางคืน อัตโนมัติตามเวลาเครื่องเซิร์ฟ
//===== คำอธิบาย ============================================
//= - ใช้นาฬิกาของเครื่องที่รัน map-server (เวลาเดียวกับ PC)
//= - ค่าเริ่มต้น: 07:00 = @day, 18:00 = @night (แก้ได้ใน RO_Manager แท็บ "เวลา / กลางวัน-กลางคืน")
//= - เปิดเซิร์ฟตอนไหนก็ได้ ระบบตั้งกลางวัน/กลางคืนให้ตรงเวลาทันที
//= - RO_Manager สั่ง "กลางวันเลย / กลางคืนเลย" ได้ ค่าที่สั่งจะอยู่ถึงรอบเวลาถัดไป
//= - ตาราง rox_daynight (ตั้งค่า + คำสั่ง), rox_daynight_state (สถานะให้ RO_Manager แสดง)
//= - ต้องปิดรอบกลางวัน/กลางคืนของเกมเอง: day_duration: 0 / night_duration: 0 (conf/battle/misc.conf)
//= - ไฟล์นี้บันทึกเป็น TIS-620
//============================================================

-	script	RoX_DayNight	-1,{
OnInit:
	query_sql "CREATE TABLE IF NOT EXISTS `rox_daynight` (`id` TINYINT NOT NULL PRIMARY KEY,`enabled` TINYINT NOT NULL DEFAULT 1,`day_min` INT NOT NULL DEFAULT 420,`night_min` INT NOT NULL DEFAULT 1080,`announce` TINYINT NOT NULL DEFAULT 0,`day_msg` VARCHAR(200) NOT NULL DEFAULT '',`night_msg` VARCHAR(200) NOT NULL DEFAULT '',`cmd` TINYINT NOT NULL DEFAULT 0,`updated` DATETIME NULL) ENGINE=MyISAM";
	query_sql "CREATE TABLE IF NOT EXISTS `rox_daynight_state` (`id` TINYINT NOT NULL PRIMARY KEY,`is_night` TINYINT NOT NULL DEFAULT 0,`server_ts` INT UNSIGNED NOT NULL DEFAULT 0,`server_time` VARCHAR(32) NOT NULL DEFAULT '',`hold` TINYINT NOT NULL DEFAULT -1,`updated` DATETIME NULL) ENGINE=MyISAM";
	query_sql "INSERT IGNORE INTO `rox_daynight` (`id`,`enabled`,`day_min`,`night_min`,`announce`,`day_msg`,`night_msg`,`cmd`,`updated`) VALUES (1,1,420,1080,0,'[Ro-X] รุ่งเช้าแล้ว ออกผจญภัยกันเถอะ!','[Ro-X] ค่ำแล้ว ระวังตัวด้วยนะ',0,NOW())";
	.hold = -1;	// ค่าที่ถูกสั่งเอง (0 กลางวัน / 1 กลางคืน) ค้างไว้จนตารางเวลาเปลี่ยนรอบ
	.hold_sched = -1;
	.first = 1;
	initnpctimer;
	end;

OnTimer5000:
	stopnpctimer;
	query_sql "SELECT `enabled`,`day_min`,`night_min`,`announce`,`day_msg`,`night_msg`,`cmd` FROM `rox_daynight` WHERE `id`=1", .@en, .@dmin, .@nmin, .@ann, .@dmsg$, .@nmsg$, .@cmd;

	// ตารางเวลา: ช่วง [day, night) = กลางวัน, นอกนั้นกลางคืน (รองรับข้ามเที่ยงคืน)
	.@now = gettime(DT_HOUR) * 60 + gettime(DT_MINUTE);
	if (.@dmin < .@nmin)
		.@sched = (.@now >= .@dmin && .@now < .@nmin) ? 0 : 1;
	else
		.@sched = (.@now >= .@nmin && .@now < .@dmin) ? 1 : 0;

	// คำสั่งจาก RO_Manager
	if (.@cmd == 1 || .@cmd == 2) {
		query_sql "UPDATE `rox_daynight` SET `cmd`=0 WHERE `id`=1";
		.hold = (.@cmd == 2) ? 1 : 0;
		.hold_sched = .@sched;
		callsub S_Set, .hold, .@ann, .@dmsg$, .@nmsg$;
	}

	// ค่าที่สั่งเองหมดอายุเมื่อถึงรอบเวลาถัดไป หรือเมื่อแก้ตั้งค่าใน RO_Manager
	.@conf$ = .@en + ":" + .@dmin + ":" + .@nmin;
	if (.hold >= 0 && (.hold_sched != .@sched || (.conf$ != "" && .conf$ != .@conf$)))
		.hold = -1;
	.conf$ = .@conf$;

	if (.@en && .hold < 0)
		callsub S_Set, .@sched, (.first ? 0 : .@ann), .@dmsg$, .@nmsg$;

	.first = 0;
	query_sql "REPLACE INTO `rox_daynight_state` (`id`,`is_night`,`server_ts`,`server_time`,`hold`,`updated`) VALUES (1," + (isnight() ? 1 : 0) + "," + gettimetick(2) + ",'" + gettimestr("%Y-%m-%d %H:%M:%S", 21) + "'," + .hold + ",NOW())";
	initnpctimer;
	end;

// S_Set <0=กลางวัน|1=กลางคืน>, <ประกาศ>, <ข้อความกลางวัน$>, <ข้อความกลางคืน$>
S_Set:
	if (getarg(0) == 1 && !isnight()) {
		night;
		if (getarg(1) && getarg(3) != "")
			announce getarg(3), bc_all, 0x9999FF;
		debugmes "[RoX DayNight] night (" + gettimestr("%H:%M", 6) + ")";
	} else if (getarg(0) == 0 && isnight()) {
		day;
		if (getarg(1) && getarg(2) != "")
			announce getarg(2), bc_all, 0xFFDD66;
		debugmes "[RoX DayNight] day (" + gettimestr("%H:%M", 6) + ")";
	}
	return;
}
